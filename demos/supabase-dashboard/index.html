Production Supabase SQL Schema & RLS Policies

Paste this directly into your Supabase SQL Editor for your GitHub project

-- 1. Enable UUID extension & create Profiles table linked to Supabase Auth
create table public.profiles (
  id uuid references auth.users on delete cascade primary key,
  full_name text not null,
  daily_calorie_goal int default 2350,
  daily_protein_goal int default 145,
  target_wake_time time default '06:15:00',
  updated_at timestamptz default now()
);

-- 2. Food Photo & Nutrition Logs Table
create table public.meal_logs (
  id uuid default gen_random_uuid() primary key,
  user_id uuid references auth.users on delete cascade not null,
  meal_name text not null,
  meal_category text not null,
  photo_url text,
  calories int not null,
  protein_g numeric(5,1) not null,
  carbs_g numeric(5,1) not null,
  fats_g numeric(5,1) not null,
  fiber_g numeric(5,1) default 0,
  eaten_at time default current_time,
  created_at timestamptz default now()
);

-- 3. Smartwatch Biometrics & Sleep Telemetry Table
create table public.watch_telemetry (
  id uuid default gen_random_uuid() primary key,
  user_id uuid references auth.users on delete cascade not null,
  device_name text not null,
  heart_rate_bpm int not null,
  hrv_ms int not null,
  spo2_pct int default 98,
  steps_count int default 0,
  active_calories_burned int default 0,
  sleep_hours numeric(4,2) default 7.5,
  recorded_at timestamptz default now()
);

-- 4. Enable Row Level Security (RLS) so users can ONLY access their own data
alter table public.profiles enable row level security;
alter table public.meal_logs enable row level security;
alter table public.watch_telemetry enable row level security;

create policy "Users manage own profile" on public.profiles
  for all using (auth.uid() = id) with check (auth.uid() = id);

create policy "Users manage own meal logs" on public.meal_logs
  for all using (auth.uid() = user_id) with check (auth.uid() = user_id);

create policy "Users manage own watch telemetry" on public.watch_telemetry
  for all using (auth.uid() = user_id) with check (auth.uid() = user_id);

-- 5. Enable Supabase Realtime (WebSockets) on watch_telemetry & meal_logs
alter publication supabase_realtime add table public.watch_telemetry;
alter publication supabase_realtime add table public.meal_logs;
