Production Supabase SQL Schema (Profiles, Meals, Beverages, Watch & Sleep)

Ready to paste into your Supabase SQL Editor

-- 1. User Profiles (Photo, Height, Weight, Age, Gender & Auto-Targets)
create table public.profiles (
  id uuid references auth.users on delete cascade primary key,
  full_name text not null,
  avatar_url text,
  gender text default 'male',
  age int default 24,
  height_cm numeric(5,1) default 178,
  weight_kg numeric(5,1) default 74,
  fitness_goal text default 'maintain',
  recommended_calories int default 2420,
  recommended_protein_g int default 148,
  recommended_water_ml int default 3100,
  recommended_sleep_hrs numeric(4,2) default 7.8,
  updated_at timestamptz default now()
);

-- 2. Meal Photo Logs
create table public.meal_logs (
  id uuid default gen_random_uuid() primary key,
  user_id uuid references auth.users on delete cascade not null,
  log_date date default current_date,
  meal_name text not null,
  meal_category text not null,
  photo_url text,
  calories int not null,
  protein_g numeric(5,1) not null,
  carbs_g numeric(5,1) not null,
  fats_g numeric(5,1) not null,
  fiber_g numeric(5,1) default 0,
  created_at timestamptz default now()
);

-- 3. Water & Beverage Logs (with Custom ml Editing & Drink Photo URL)
create table public.beverage_logs (
  id uuid default gen_random_uuid() primary key,
  user_id uuid references auth.users on delete cascade not null,
  log_date date default current_date,
  beverage_name text not null,
  volume_ml int not null,
  calories int default 0,
  protein_g numeric(5,1) default 0,
  carbs_g numeric(5,1) default 0,
  photo_url text,
  created_at timestamptz default now()
);

-- 4. Daily Smartwatch & Sleep History (Stored Every Day)
create table public.daily_watch_sleep_logs (
  id uuid default gen_random_uuid() primary key,
  user_id uuid references auth.users on delete cascade not null,
  log_date date default current_date,
  watch_device text,
  steps_count int default 0,
  avg_bpm int default 68,
  active_burn_kcal int default 0,
  sleep_hours numeric(4,2) default 7.5,
  total_calories_eaten int default 0,
  total_protein_g int default 0,
  total_fluid_ml int default 0,
  unique(user_id, log_date)
);

alter table public.profiles enable row level security;
alter table public.meal_logs enable row level security;
alter table public.beverage_logs enable row level security;
alter table public.daily_watch_sleep_logs enable row level security;

create policy "Users own profile" on public.profiles for all using (auth.uid() = id);
create policy "Users own meals" on public.meal_logs for all using (auth.uid() = user_id);
create policy "Users own beverages" on public.beverage_logs for all using (auth.uid() = user_id);
create policy "Users own daily watch sleep" on public.daily_watch_sleep_logs for all using (auth.uid() = user_id);
